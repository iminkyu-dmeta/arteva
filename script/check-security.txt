4.1.2 패스워드 복잡성 설정
# 숫자 최소 사용 수 설정 (dcredit = -1)
cat /etc/security/pwquality.conf | grep dcredit

# 이전에 사용하던 패스워드에서 사용한 문자 허용 금지 수 설정(difok = 4)"
cat /etc/security/pwquality.conf | grep difok

# 소문자 최소 사용 수 설정(lcredit = -1)
cat /etc/security/pwquality.conf | grep lcredit

# 동일한 타입(대소문자, 숫자, 특수문자)의 최대 허용 연속 문자 수 설정(maxclassrepeat = 4)
cat /etc/security/pwquality.conf | grep maxclassrepeat

# 동일 문자 사용 최대 허용 문자 수 설정(maxrepeat = 3)
cat /etc/security/pwquality.conf | grep maxrepeat
 
# 패스워드 최소 길이 설정(minlen = 12)
cat /etc/security/pwquality.conf | grep minlen

# 특수문자 최소 사용 수 설정(ocredit = -1)
cat /etc/security/pwquality.conf | grep ocredit
 
# 대문자 최소 사용 수 설정(ucredit = -1)
cat /etc/security/pwquality.conf | grep ucredit


4.1.3 계정 잠금 임계값 설정
# 패스워드 입력 실패시 잠금 (deny =3)

# 900 초 동안 연속 인증 실패시 계정 잠김(fail_interval = 900)

# 600 초 후 계정 해제 (unlock_time = 600)
cat /etc/security/faillock.conf | grep '^unlock_time'

4.1.15 파일 및 디렉터리 소유자 설정
# /etc/passwd 파일 소유자  및  권한 설정(644)

# /etc/shadow 파일 소유자 및 권한 설정(400)

# /etc/hosts 파일 소유자 및 권한 설정(644)
ls -l /etc/hosts
# /etc/(x)inetd.conf 파일 소유자 및 권한 설정 (미사용) 
ls -al /etc/xinetd.d/
# /etc/syslog.conf 파일 소유자  및 권한 설정 (미사용)
ls –al /etc/syslog.conf
# /etc/services 파일 소유자 및 권한 설정(644)
ls -al /etc/services


4.1.16 SUID, SGID, Sticky bit 설정 파일 점검
# /apps 파일 존재여부
sudo find / -xdev -user root -type f \( -perm -04000 -o -perm -02000 \) -exec ls -al {} \;

# r 계열 서비스 비활성화 (rsh 등등 실행 process 없음)

# cron 파일 소유자 및 권한설정 (미사용 /etc/cron.deny)
ls -al /etc/cron.deny
# Dos 공격에 취약한 서비스 비활성화 (실행 Process 없음)
sudo ls -alL /etc/xinetd.d/* | egrep '(echo|discard|daytime|charge)‘
# NFS 서비스 비활성화, NFS 접근 통제 (실행 데몬 없음)
ps -ef | egrep "nfsd|statd|mountd"

4.2.2 기본 계정의 패스워드, 권한 등을 변경하여 사용

sudo mariadb -u root -p mysql -t -e "select User, Password from user where User = 'root';"


4.2.3 데이터베이스의 불필요 계정을 제거하거나, 잠금설정 후 사용

4.2.4 패스워드의 사용기간 및 복잡도를 기관 정책에 맞도록 설정

sudo mariadb -u root -p mysql -t -e "select Host, User, password_expired, plugin from user where Host = 'localhost';"
# password  정책 상태
sudo mariadb -u root -p mysql -t -e "show variables like 'strict_password_validation';"
